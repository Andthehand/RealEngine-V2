31 void VertexArray::SetVertexBuffers(std::vector<Ref<VertexBuffer>> vertexBuffers) {
32 RE_PROFILE_FUNCTION();
33 m_VertexBuffer = vertexBuffers;
34
35 Bind();
36 uint8_t index = 0; // Since there are multiple VBO we need to keep track of the what the gloabal index is
37 for (const Ref<VertexBuffer> vertexBuffer : vertexBuffers) {
38 vertexBuffer->Bind();
39
40 const BufferAttributes& layout = vertexBuffer->GetLayout();
41 const std::vector<BufferAttribute>& attribs = layout.m_VertexAttribs;
42 for (uint8_t i = 0; i < attribs.size(); i++) {
43 const BufferAttribute& attrib = attribs[i];
44
45 switch (attrib.Type.GetType()) {
46 case DataType::Float:
47 case DataType::Float2:
48 case DataType::Float3:
49 case DataType::Float4:
50 glVertexAttribPointer(index, attrib.Size, attrib.Type.GetGLType(), GL_FALSE, layout.Stride, (void*)(uintptr_t)attrib.Offset);
51 break;
52 case DataType::Uint:
53 case DataType::Uint2:
54 case DataType::Uint3:
55 case DataType::Uint4:
56 glVertexAttribIPointer(index, attrib.Size, attrib.Type.GetGLType(), layout.Stride, (void*)(uintptr_t)attrib.Offset);
57 break;
58 default:
59 RE_CORE_ASSERT(false, "DataType not supported yet!");
60 break;
61 }
62 glVertexAttribDivisor(index, attrib.InstanceDivisor);
63 glEnableVertexAttribArray(index);
64 index++;
65 }
66 }
