41 void Framebuffer::Invalidate() {
42 RE_PROFILE_FUNCTION();
43
44 if (m_FramebufferID) {
45 glDeleteFramebuffers(1, &m_FramebufferID);
46 glDeleteTextures((GLsizei)m_Attachments.size(), m_Attachments.data());
47 }
48
49 glCreateFramebuffers(1, &m_FramebufferID);
50 glBindFramebuffer(GL_FRAMEBUFFER, m_FramebufferID);
51
52 for (uint32_t i = 0; i < m_Attachments.size(); i++) {
53 auto& attachment = m_Specification.Attachments.Attachments[i];
54
55 glCreateTextures(GL_TEXTURE_2D, 1, &m_Attachments[i]);
56 glBindTexture(GL_TEXTURE_2D, m_Attachments[i]);
57
58 switch (attachment.TextureFormat) {
59 case FramebufferTextureFormat::RGBA8:
60 glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, m_Specification.Width, m_Specification.Height, 0, GL_RGBA, GL_UNSIGNED_BYTE, nullptr);
61 break;
62 case FramebufferTextureFormat::RED_INTEGER:
63 glTexImage2D(GL_TEXTURE_2D, 0, GL_RED_INTEGER, m_Specification.Width, m_Specification.Height, 0, GL_RED_INTEGER, GL_UNSIGNED_BYTE, nullptr);
64 break;
65 case FramebufferTextureFormat::DEPTH24_STENCIL8:
66 glTexStorage2D(GL_TEXTURE_2D, 1, GL_DEPTH24_STENCIL8, m_Specification.Width, m_Specification.Height);
67 break;
68 default:
69 RE_CORE_ASSERT(false, "Unknown texture format");
70 }
71
72 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
73 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
74 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_R, GL_CLAMP_TO_EDGE);
75 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
76 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
77
78 if (attachment.TextureFormat != FramebufferTextureFormat::DEPTH24_STENCIL8) {
79 glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, m_Attachments[i], 0);
80 }
81 else {
82 glFramebufferTexture2D(GL_FRAMEBUFFER, GL_DEPTH_STENCIL_ATTACHMENT, GL_TEXTURE_2D, m_Attachments[i], 0);
83 }
84
85 RE_CORE_ASSERT(m_Attachments.size() <= 4);
86 GLenum buffers[4] = { GL_COLOR_ATTACHMENT0, GL_COLOR_ATTACHMENT1, GL_COLOR_ATTACHMENT2, GL_COLOR_ATTACHMENT3 };
87 glDrawBuffers((GLsizei)m_Attachments.size(), buffers);
88
89 RE_CORE_ASSERT(glCheckFramebufferStatus(GL_FRAMEBUFFER) == GL_FRAMEBUFFER_COMPLETE, "Framebuffer is incomplete!");
90
91 glBindFramebuffer(GL_FRAMEBUFFER, 0);
92 }
93 }
