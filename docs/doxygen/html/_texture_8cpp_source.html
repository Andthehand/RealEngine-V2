75 void Texture2DArray::LoadTextures(It begin, uint32_t numTextures, uint32_t mipLevels) {
76 RE_PROFILE_FUNCTION();
77
78 RE_CORE_ASSERT(numTextures != 0, "Can't make Texuture without any textures");
79 TextureData* textureData = new TextureData[numTextures];
80 stbi_uc** data = new stbi_uc * [numTextures];
81
82 int checkChannel = -1;
83 stbi_set_flip_vertically_on_load(1);
84 //Load all of the images into memory
85 for (uint32_t i = 0; i < numTextures; i++) {
86 const std::filesystem::path& path = *(begin + i);
87
88 data[i] = stbi_load(path.string().c_str(), &textureData[i].Width, &textureData[i].Height, &textureData[i].Channels, 0);
89 RE_CORE_ASSERT(data, "Failed to load image with path: {}", path);
90
91 //Set data only on the first texture upload
92 if (checkChannel == -1) {
93 checkChannel = textureData[i].Channels;
94 m_Width = textureData[i].Width;
95 m_Height = textureData[i].Height;
96 }
97
98 //Make sure the channels, width, and height are the same
99 RE_CORE_ASSERT(checkChannel == textureData[i].Channels &&
100 m_Width == (uint32_t)textureData[i].Width &&
101 m_Height == (uint32_t)textureData[i].Height,
102 "Channel, Width, or height do not match while creating TextureArray");
103 }
104
105 GLenum internalFormat = 0, dataFormat = 0;
106 switch (checkChannel) {
107 case 3: internalFormat = GL_RGB8; dataFormat = GL_RGB; break;
108 case 4: internalFormat = GL_RGBA8; dataFormat = GL_RGBA; break;
109 default: RE_CORE_ASSERT(false, "Grayscale Images are not supported!");
110 }
111
112 //Create TextureArray and allocate memory
113 glCreateTextures(GL_TEXTURE_2D_ARRAY, 1, &m_RendererID);
114 glTextureStorage3D(m_RendererID, mipLevels, internalFormat, m_Width, m_Height, numTextures);
115 //Upload the images to the GPU
116 for (uint32_t i = 0; i < numTextures; i++) {
117 glTextureSubImage3D(m_RendererID, 0, 0, 0, i, m_Width, m_Height, 1, dataFormat, GL_UNSIGNED_BYTE, data[i]);
118 }
119 glGenerateTextureMipmap(m_RendererID);
120
121 GLfloat max_aniso;
122 glGetFloatv(GL_MAX_TEXTURE_MAX_ANISOTROPY, &max_aniso);
123 glTextureParameterf(m_RendererID, GL_TEXTURE_MAX_ANISOTROPY, max_aniso);
124 glTextureParameteri(m_RendererID, GL_TEXTURE_MIN_FILTER, GL_NEAREST_MIPMAP_NEAREST);
125 glTextureParameteri(m_RendererID, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
126 glTextureParameteri(m_RendererID, GL_TEXTURE_WRAP_S, GL_REPEAT);
127 glTextureParameteri(m_RendererID, GL_TEXTURE_WRAP_T, GL_REPEAT);
128
129 //Delete rescources
130 for (uint32_t i = 0; i < numTextures; i++) {
131 stbi_image_free(data[i]);
132 }
133 delete[] data;
134 delete[] textureData;
135 }
