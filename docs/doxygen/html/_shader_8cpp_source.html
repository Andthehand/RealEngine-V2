77 void Shader::CompileShader(const std::vector<ShaderProcessing>& shaderProcessing) {
78 RE_PROFILE_FUNCTION();
79 std::vector<uint32_t> shaderIDs;
80 for (const ShaderProcessing& shaderCode : shaderProcessing) {
81 shaderIDs.push_back(glCreateShader(Utils::ShaderTypeToGLType(shaderCode.ShaderType)));
82
83 const char* shaderSource = shaderCode.ShaderCode.c_str();
84 glShaderSource(shaderIDs.back(), 1, &shaderSource, NULL);
85 glCompileShader(shaderIDs.back());
86
87 //Check for errors
88 int success;
89 glGetShaderiv(shaderIDs.back(), GL_COMPILE_STATUS, &success);
90 if (!success) {
91 int maxLength = 0;
92 glGetShaderiv(shaderIDs.back(), GL_INFO_LOG_LENGTH, &maxLength);
93
94 std::vector<char> infoLog(maxLength);
95 glGetShaderInfoLog(shaderIDs.back(), maxLength, &maxLength, &infoLog[0]);
96
97 glGetShaderInfoLog(shaderIDs.back(), 512, NULL, &infoLog[0]);
98 RE_CORE_ERROR("{0} shader compilation failed: {1}", Utils::ShaderTypeToString(shaderCode.ShaderType), infoLog.data());
99 }
100 }
101
102 // shader Program
103 m_ShaderID = glCreateProgram();
104 for (const uint32_t id : shaderIDs) {
105 glAttachShader(m_ShaderID, id);
106 }
107 glLinkProgram(m_ShaderID);
108
109 //Check for errors
110 int success;
111 glGetProgramiv(m_ShaderID, GL_LINK_STATUS, &success);
112 if (!success) {
113 int maxLength = 0;
114 glGetProgramiv(m_ShaderID, GL_INFO_LOG_LENGTH, &maxLength);
115
116 std::vector<char> infoLog(maxLength);
117 glGetProgramInfoLog(m_ShaderID, maxLength, &maxLength, &infoLog[0]);
118 RE_CORE_ERROR("Shader program linking failed: {0}", infoLog.data());
119
120 glDeleteProgram(m_ShaderID);
121 }
122
123 // delete the shaders as they're linked into our program now and no longer necessary
124 for (const uint32_t id : shaderIDs) {
125 glDeleteShader(id);
126 }

129 void Shader::ParseFile(const std::filesystem::path& file) {
130 RE_PROFILE_FUNCTION();
131 std::ifstream stream(file);
132
133 if (!stream.is_open()) {
134 RE_CORE_ERROR("Failed to open file: {0}", file.string());
135 return;
136 }
137
138 // Read in file to string
139 std::string fileString((std::istreambuf_iterator<char>(stream)), std::istreambuf_iterator<char>());
140
141 std::vector<ShaderProcessing> shaderProcessing;
142 size_t offset = fileString.find("#type");
143 RE_CORE_ASSERT(offset != std::string::npos, "No shader code found in file {0}", file);
144 while (offset != std::string::npos) {
145 offset = fileString.find_first_not_of(' ', offset += 5); // Jump to first non whitespace after #type hence += 5
146 size_t endOfShaderType = fileString.find_first_of('\n', offset);
147
148 std::string_view shaderTypeString = std::string_view(fileString.data() + offset, endOfShaderType - offset);
149 ShaderTypes shaderType = Utils::CheckShaderTypeString(shaderTypeString);
150
151 offset = fileString.find("#type", offset); // Get position of next shader code
152 shaderProcessing.push_back({ fileString.substr(endOfShaderType, offset - endOfShaderType), shaderType});
153 }
154
155 CompileShader(shaderProcessing);
156 }
